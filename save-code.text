import {
  DropdownMenu,
  DropdownMenuCheckboxItem,
  DropdownMenuContent,
  DropdownMenuGroup,
  DropdownMenuItem,
  DropdownMenuLabel,
  DropdownMenuTrigger,
} from "@/components/ui/dropdown-menu";
import { cn } from "@/lib/utils";
import {
  BarChart3,
  Calendar,
  CheckSquare,
  FileText,
  FolderKanban,
  LayoutDashboard,
  LucideIcon,
  Pin,
  Settings,
  Settings2,
  Users,
} from "lucide-react";
import { useState } from "react";
import { Button } from "../ui/button";

type NavItem = {
  key: string;
  label: string;
  icon: LucideIcon;
  priority: number;
  state: navItemState;
};

type navItemState = "selected" | "firstPeneded" | "secondPended" | "inMore";
const NAV_ITEMS: NavItem[] = [
  {
    key: "dashboard",
    label: "Dashboard",
    icon: LayoutDashboard,
    priority: 1,
    state: "selected",
  },
  {
    key: "tasks",
    label: "Tasks",
    icon: CheckSquare,
    priority: 2,
    state: "firstPeneded",
  },
  {
    key: "projects",
    label: "Projects",
    icon: FolderKanban,
    priority: 3,
    state: "secondPended",
  },
  {
    key: "calendar",
    label: "Calendar",
    icon: Calendar,
    priority: 4,
    state: "inMore",
  },
  { key: "team", label: "Team", icon: Users, priority: 5, state: "inMore" },
  {
    key: "analytics",
    label: "Analytics",
    icon: BarChart3,
    priority: 6,
    state: "inMore",
  },
  {
    key: "files",
    label: "Files",
    icon: FileText,
    priority: 7,
    state: "inMore",
  },
  {
    key: "settings",
    label: "Settings",
    icon: Settings,
    priority: 8,
    state: "inMore",
  },
];

export default function NavSwitcher() {
  const [navItems, setNavItems] = useState(NAV_ITEMS);

  const setSelected = (item: NavItem, alreadySelected: NavItem) => {
    const newItems: NavItem[] = navItems.map((e) => {
      if (e.key === item.key) {
        console.log(1);

        return {
          ...e,
          state: "selected",
        };
      } else if (e.key === alreadySelected.key) {
        console.log(2);
        return {
          ...e,
          state: item.state,
        };
      } else {
        console.log(3);
        return e;
      }
    });

    setNavItems(newItems);
  };
  const setFirstPended = (item: NavItem, alreadyFirstPended: NavItem) => {
    const newItems: NavItem[] = navItems.map((e) => {
      if (e.key === item.key) {
        return {
          ...e,
          state: "firstPeneded",
        };
      } else if (e.key === alreadyFirstPended.key) {
        return {
          ...e,
          state: item.state,
        };
      } else {
        return e;
      }
    });
    setNavItems(newItems);
  };
  const setSecondPended = (item: NavItem, alreadyFirstPended: NavItem) => {
    const newItems: NavItem[] = navItems.map((e) => {
      if (e.key === item.key) {
        return {
          ...e,
          state: "secondPended",
        };
      } else if (e.key === alreadyFirstPended.key) {
        return {
          ...e,
          state: item.state,
        };
      } else {
        return e;
      }
    });
    setNavItems(newItems);
  };

  const selectedItem = navItems.find((item) => item.state === "selected")!;
  const firstPendedItem = navItems.find(
    (item) => item.state === "firstPeneded",
  )!;
  const secondPendedItem = navItems.find(
    (item) => item.state === "secondPended",
  )!;
  const inMoreItems = navItems.filter((item) => item.state === "inMore");

  return (
    <div className="flex gap-3 items-center justify-center py-2 border rounded-md px-2 mt-2">
      {selectedItem &&
        firstPendedItem &&
        secondPendedItem &&
        [selectedItem, firstPendedItem, secondPendedItem].map((item) => {
          return (
            <Button
              key={item.key}
              size="icon"
              variant={false ? "outline" : "ghost"}
              className={cn(
                "size-10 flex-col hover:text-primary hover:bg-transparent transform-content cursor-pointer",
                {
                  "text-primary bg-primary/10 hover:bg-primary/10 border-primary size-14":
                    item.state === "selected",
                },
              )}
              onClick={() => setSelected(item, selectedItem)}
            >
              <item.icon className={cn("size-4", {
                "size-6": item.state === 'selected'
              })} />
              <span className="text text-xxs font-bold">{item.key}</span>
            </Button>
          );
        })}
      <DropdownMenu>
        <DropdownMenuTrigger asChild>
          <Button size="icon" variant="outline" className="size-10 flex-col">
            <Settings2 className="size-4" />
            <span className="text text-xxs font-bold">Custm</span>
          </Button>
        </DropdownMenuTrigger>
        <DropdownMenuContent className="w-48">
          <DropdownMenuGroup>
            <DropdownMenuLabel>Sections</DropdownMenuLabel>
            {inMoreItems.map((v) => {
              return (
                <DropdownMenuItem
                  key={v.key}
                  onSelect={() => setSelected(v, selectedItem)}
                >
                  <v.icon />
                  <span className="flex-1">{v.key}</span>
                  <div
                    className="gruop p-1.5 cursor-pointer group relative"
                    onClick={(e) => {
                      e.stopPropagation();
                      setFirstPended(v, firstPendedItem);
                    }}
                  >
                    <Pin className="size-3 self-end  group-hover:fill-muted-foreground group-hover:stroke-muted-foreground" />
                    <span className="absolute text-xxs size-3 flex justify-center in-checked: rounded-full top-0 left-0">
                      1
                    </span>
                  </div>
                  <div
                    className="gruop p-1.5 cursor-pointer group relative"
                    onClick={(e) => {
                      e.stopPropagation();
                      setSecondPended(v, secondPendedItem);
                    }}
                  >
                    <Pin className="size-3 self-end  group-hover:fill-muted-foreground group-hover:stroke-muted-foreground" />
                    <span className="absolute text-xxs size-3 flex justify-center in-checked: rounded-full top-0 left-0">
                      2
                    </span>
                  </div>
                </DropdownMenuItem>
              );
            })}
          </DropdownMenuGroup>
        </DropdownMenuContent>
      </DropdownMenu>
    </div>
  );
}
